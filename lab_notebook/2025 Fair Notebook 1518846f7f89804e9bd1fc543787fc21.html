    def preprocess_image(self, image): # this adjusts the image to dimensions (1, 128, 128, 128) for UNet compatability using trilinear interpolation (order=3)
        # resample image to (128, 128, 128)
        zoom_factors = (
            128 / image.shape[0],
            128 / image.shape[1],
            128 / image.shape[2]
        )
        image = zoom(image, zoom_factors, order=3)
        image = (image - np.min(image)) / (np.max(image) - np.min(image) + 1e-8)
        image = 2 * image - 1  # normalize to [-1, 1]
        return np.expand_dims(image, axis=0)  # add channel dimension (grayscale value on [0,1])

    def preprocess_mask(self, mask): # this adjusts the mask to dimensions (4, 128, 128, 128) for UNet compatability using nearest-neighbor interpolation (order=0)
        # resample mask to (128, 128, 128)
        zoom_factors = (
            128 / mask.shape[0],
            128 / mask.shape[1],
            128 / mask.shape[2]
        )
        mask = zoom(mask, zoom_factors, order=0)
        # one-hot encode the mask, labels from 1 to 4
        num_classes = 4

Next, I implement sinusoidal positional embedding to allow the model to use the time step ttt﻿ as conditioning as described in https://arxiv.org/pdf/2006.11239 by encoding ttt﻿ into a high-dimensional space using sinusoidal functions:

# sinusoidal positional embedding
class SinusoidalPositionEmbeddings(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.dim = dim

    def forward(self, t):
        device = t.device
        half_dim = self.dim // 2
        emb = -np.log(10000) / (half_dim - 1)
        emb = torch.exp(torch.arange(half_dim, device=device) * emb)
        emb = t.float()[:, None] * emb[None, :]
        emb = torch.cat([torch.sin(emb), torch.cos(emb)], dim=-1)
        return emb

# residual block with optional attention mechanism
class ResidualBlock(nn.Module):
    def __init__(self, in_channels, out_channels, time_emb_dim, use_attention=False):
        super().__init__()
        self.time_mlp = nn.Sequential(
            nn.SiLU(),
            nn.Linear(time_emb_dim, out_channels)
        )
        self.block1 = nn.Sequential(
            nn.Conv3d(in_channels, out_channels, 3, padding=1),
            nn.GroupNorm(num_groups=8, num_channels=out_channels),
            nn.SiLU()
        )
        if use_attention:
            self.attention = SelfAttention3D(out_channels)
        else:
            self.attention = nn.Identity()
        self.block2 = nn.Sequential(
            nn.Conv3d(out_channels, out_channels, 3, padding=1),
            nn.GroupNorm(num_groups=8, num_channels=out_channels),
            nn.SiLU()
        )
        if in_channels != out_channels:
            self.res_conv = nn.Conv3d(in_channels, out_channels, 1)
        else:
            self.res_conv = nn.Identity()

    def forward(self, x, t):
        h = self.block1(x)
        time_emb = self.time_mlp(t)
        time_emb = time_emb[..., None, None, None]
        h = h + time_emb
        h = self.attention(h)
        h = self.block2(h)
        return h + self.res_conv(x)

# self-attention layer
class SelfAttention3D(nn.Module):
    def __init__(self, channels):
        super().__init__()
        self.group_norm = nn.GroupNorm(num_groups=8, num_channels=channels)
        self.q = nn.Conv3d(channels, channels, 1)
        self.k = nn.Conv3d(channels, channels, 1)
        self.v = nn.Conv3d(channels, channels, 1)
        self.proj_out = nn.Conv3d(channels, channels, 1)

    def forward(self, x):
        B, C, H, W, D = x.shape
        x_norm = self.group_norm(x)
        q = self.q(x_norm).reshape(B, C, -1)  # [B, C, N]
        k = self.k(x_norm).reshape(B, C, -1)  # [B, C, N]
        v = self.v(x_norm).reshape(B, C, -1)  # [B, C, N]
        attn = torch.bmm(q.transpose(1, 2), k) / np.sqrt(C)  # [B, N, N]
        attn = torch.softmax(attn, dim=-1)  # [B, N, N]
        h = torch.bmm(attn, v.transpose(1, 2))  # [B, N, C]
        h = h.transpose(1, 2).reshape(B, C, H, W, D)  # [B, C, H, W, D]
        h = self.proj_out(h)  # [B, C, H, W, D]
        return x + h  # Residual connection

# unet3d model
class UNet3D(nn.Module):
    def __init__(self, in_channels, out_channels, time_emb_dim=128, initial_features=64):
        super().__init__()
        self.time_emb_dim = time_emb_dim
        self.time_emb = SinusoidalPositionEmbeddings(time_emb_dim)
        self.time_mlp = nn.Sequential(
            nn.Linear(time_emb_dim, time_emb_dim * 4),
            nn.SiLU(),
            nn.Linear(time_emb_dim * 4, time_emb_dim)
        )
        features = initial_features
        # encoder
        self.enc1 = ResidualBlock(in_channels, features, time_emb_dim)
        self.pool1 = nn.MaxPool3d(2)
        self.enc2 = ResidualBlock(features, features * 2, time_emb_dim)
        self.pool2 = nn.MaxPool3d(2)
        self.enc3 = ResidualBlock(features * 2, features * 4, time_emb_dim, use_attention=True)
        self.pool3 = nn.MaxPool3d(2)
        self.enc4 = ResidualBlock(features * 4, features * 8, time_emb_dim, use_attention=True)
        self.pool4 = nn.MaxPool3d(2)
        # bottleneck
        self.bottleneck = ResidualBlock(features * 8, features * 8, time_emb_dim, use_attention=True)
        # decoder
        self.upconv4 = nn.ConvTranspose3d(features * 8, features * 8, kernel_size=2, stride=2)
        self.dec4 = ResidualBlock(features * 8 * 2, features * 8, time_emb_dim, use_attention=True)
        self.upconv3 = nn.ConvTranspose3d(features * 8, features * 4, kernel_size=2, stride=2)
        self.dec3 = ResidualBlock(features * 4 * 2, features * 4, time_emb_dim, use_attention=True)
        self.upconv2 = nn.ConvTranspose3d(features * 4, features * 2, kernel_size=2, stride=2)
        self.dec2 = ResidualBlock(features * 2 * 2, features * 2, time_emb_dim)
        self.upconv1 = nn.ConvTranspose3d(features * 2, features, kernel_size=2, stride=2)
        self.dec1 = ResidualBlock(features * 2, features, time_emb_dim)
        # output
        self.conv = nn.Conv3d(features, out_channels, kernel_size=1)

    def forward(self, x, t):
        # time embedding
        t_emb = self.time_emb(t)
        t_emb = self.time_mlp(t_emb)
        # encoder
        enc1 = self.enc1(x, t_emb)
        enc2 = self.enc2(self.pool1(enc1), t_emb)
        enc3 = self.enc3(self.pool2(enc2), t_emb)
        enc4 = self.enc4(self.pool3(enc3), t_emb)
        # bottleneck
        bottleneck = self.bottleneck(self.pool4(enc4), t_emb)
        # decoder
        dec4 = self.upconv4(bottleneck)
        dec4 = torch.cat([dec4, enc4], dim=1)
        dec4 = self.dec4(dec4, t_emb)
        dec3 = self.upconv3(dec4)
        dec3 = torch.cat([dec3, enc3], dim=1)
        dec3 = self.dec3(dec3, t_emb)
        dec2 = self.upconv2(dec3)
        dec2 = torch.cat([dec2, enc2], dim=1)
        dec2 = self.dec2(dec2, t_emb)
        dec1 = self.upconv1(dec2)
        dec1 = torch.cat([dec1, enc1], dim=1)
        dec1 = self.dec1(dec1, t_emb)
        return self.conv(dec1)

class OneHotMask:
    def __init__(self, num_classes):
        self.num_classes = num_classes

    def __call__(self, mask):
        if isinstance(mask, np.ndarray): 
            mask = torch.from_numpy(mask).long()
        else: 
            mask = mask.long()

        # apply one-hot encoding
        one_hot = F.one_hot(mask, num_classes=self.num_classes + 1)
        one_hot = one_hot[..., 1:].permute(3, 0, 1, 2).float()  # [C, H, W, D]
        return one_hot

class EMA():
    def __init__(self, beta):
        super().__init__()
        self.beta = beta

    def update_model_average(self, ma_model, current_model):
        for current_params, ma_params in zip(current_model.parameters(), ma_model.parameters()):
            old_weight, up_weight = ma_params.data, current_params.data
            ma_params.data = self.update_average(old_weight, up_weight)

    def update_average(self, old, new):
        if old is None:
            return new
        return old * self.beta + (1 - self.beta) * new

def extract(a, t, x_shape):
    b, *_ = t.shape
    out = a.gather(-1, t)
    return out.reshape(b, *((1,) * (len(x_shape) - 1)))

def noise_like(shape, device, repeat=False):
    repeat_noise = lambda: torch.randn((1, *shape[1:]), device=device).repeat(shape[0], *((1,) * (len(shape) - 1)))
    noise = lambda: torch.randn(shape, device=device)
    return repeat_noise() if repeat else noise()

def cosine_beta_schedule(timesteps, s = 0.008):
    """
    Cosine schedule from https://openreview.net/forum?id=-NEXDKk8gZ
    """
    steps = timesteps + 1
    x = np.linspace(0, steps, steps)
    alphas_cumprod = np.cos(((x / steps) + s) / (1 + s) * np.pi * 0.5) ** 2
    alphas_cumprod = alphas_cumprod / alphas_cumprod[0]
    betas = 1 - (alphas_cumprod[1:] / alphas_cumprod[:-1])
    return np.clip(betas, a_min = 0, a_max = 0.999)

class GaussianDiffusion(nn.Module):
    def __init__(
        self,
        denoise_fn,
        *,
        image_size,
        depth_size,
        channels = 1,
        timesteps = 250,
        loss_type = 'l1',
        betas = None,
        with_condition = False,
        with_pairwised = False,
        apply_bce = False,
        lambda_bce = 0.0
    ):
        super().__init__()
        self.channels = channels
        self.image_size = image_size
        self.depth_size = depth_size
        self.denoise_fn = denoise_fn
        self.with_condition = with_condition
        self.with_pairwised = with_pairwised
        self.apply_bce = apply_bce
        self.lambda_bce = lambda_bce

        if exists(betas):
            betas = betas.detach().cpu().numpy() if isinstance(betas, torch.Tensor) else betas
        else:
            betas = cosine_beta_schedule(timesteps)

        alphas = 1. - betas
        alphas_cumprod = np.cumprod(alphas, axis=0)
        alphas_cumprod_prev = np.append(1., alphas_cumprod[:-1])

        timesteps, = betas.shape
        self.num_timesteps = int(timesteps)
        self.loss_type = loss_type

        to_torch = partial(torch.tensor, dtype=torch.float32)

        self.register_buffer('betas', to_torch(betas))
        self.register_buffer('alphas', to_torch(alphas))
        self.register_buffer('alphas_cumprod', to_torch(alphas_cumprod))
        self.register_buffer('alphas_cumprod_prev', to_torch(alphas_cumprod_prev))

        # q(x_t | x_{t-1})
        self.register_buffer('sqrt_alphas_cumprod', to_torch(np.sqrt(alphas_cumprod)))
        self.register_buffer('sqrt_one_minus_alphas_cumprod', to_torch(np.sqrt(1. - alphas_cumprod)))
        self.register_buffer('log_one_minus_alphas_cumprod', to_torch(np.log(1. - alphas_cumprod)))
        self.register_buffer('sqrt_recip_alphas_cumprod', to_torch(np.sqrt(1. / alphas_cumprod)))
        self.register_buffer('sqrt_recipm1_alphas_cumprod', to_torch(np.sqrt(1. / alphas_cumprod - 1)))

        # for q(x_{t-1} | x_t, x_0)
        posterior_variance = betas * (1. - alphas_cumprod_prev) / (1. - alphas_cumprod)
        self.register_buffer('posterior_variance', to_torch(posterior_variance))
        self.register_buffer('posterior_log_variance_clipped', to_torch(np.log(np.maximum(posterior_variance, 1e-20))))
        self.register_buffer('posterior_mean_coef1', to_torch(
            betas * np.sqrt(alphas_cumprod_prev) / (1. - alphas_cumprod)))
        self.register_buffer('posterior_mean_coef2', to_torch(
            (1. - alphas_cumprod_prev) * np.sqrt(alphas) / (1. - alphas_cumprod)))
        self.register_buffer('posterior_mean_coef3', to_torch(
            1. - (1. - alphas_cumprod_prev) * np.sqrt(alphas) / (1. - alphas_cumprod)))

    def q_mean_variance(self, x_start, t, c=None):
        x_hat = 0
        mean = extract(self.sqrt_alphas_cumprod, t, x_start.shape) * x_start + x_hat
        variance = extract(1. - self.alphas_cumprod, t, x_start.shape)
        log_variance = extract(self.log_one_minus_alphas_cumprod, t, x_start.shape)
        return mean, variance, log_variance

    def predict_start_from_noise(self, x_t, t, noise, c=None):
        x_hat = 0.
        return (
            extract(self.sqrt_recip_alphas_cumprod, t, x_t.shape) * x_t -
            extract(self.sqrt_recipm1_alphas_cumprod, t, x_t.shape) * noise -
            extract(self.sqrt_recip_alphas_cumprod, t, x_t.shape) * x_hat
        )

    def q_posterior(self, x_start, x_t, t, c=None):
        x_hat = 0.
        posterior_mean = (
            extract(self.posterior_mean_coef1, t, x_t.shape) * x_start +
            extract(self.posterior_mean_coef2, t, x_t.shape) * x_t +
            extract(self.posterior_mean_coef3, t, x_t.shape) * x_hat
        )
        posterior_variance = extract(self.posterior_variance, t, x_t.shape)
        posterior_log_variance_clipped = extract(self.posterior_log_variance_clipped, t, x_t.shape)
        return posterior_mean, posterior_variance, posterior_log_variance_clipped

    def p_mean_variance(self, x, t, clip_denoised: bool, c=None):
        if self.with_condition:
            noise_pred = self.denoise_fn(torch.cat([x, c], 1), t)
            x_recon = self.predict_start_from_noise(x, t, noise_pred, c)
        else:
            noise_pred = self.denoise_fn(x, t)
            x_recon = self.predict_start_from_noise(x, t, noise_pred, c)

        if clip_denoised:
            x_recon.clamp_(-1., 1.)

        model_mean, posterior_variance, posterior_log_variance = self.q_posterior(
            x_start=x_recon, x_t=x, t=t, c=c
        )
        return model_mean, posterior_variance, posterior_log_variance

    @torch.no_grad()
    def p_sample(self, x, t, condition_tensors=None, clip_denoised=True, repeat_noise=False):
        b, *_, device = *x.shape, x.device
        model_mean, _, model_log_variance = self.p_mean_variance(
            x=x, t=t, c=condition_tensors, clip_denoised=clip_denoised
        )
        noise = noise_like(x.shape, device, repeat_noise)
        nonzero_mask = (1 - (t == 0).float()).reshape(
            b, *((1,) * (len(x.shape) - 1))
        )
        return model_mean + nonzero_mask * (0.5 * model_log_variance).exp() * noise

    @torch.no_grad()
    def p_sample_loop(self, shape, condition_tensors=None):
        device = self.betas.device
        b = shape[0]
        img = torch.randn(shape, device=device)

        for i in tqdm(reversed(range(0, self.num_timesteps)), desc='sampling loop time step', total=self.num_timesteps):
            t = torch.full((b,), i, device=device, dtype=torch.long)
            if self.with_condition:
                img = self.p_sample(img, t, condition_tensors=condition_tensors)
            else:
                img = self.p_sample(img, t)
        return img

    @torch.no_grad()
    def sample(self, batch_size=2, condition_tensors=None):
        image_size = self.image_size
        depth_size = self.depth_size
        channels = self.channels

        # Changed the order to [H, W, D] to preserve orientation
        return self.p_sample_loop(
            (batch_size, channels, image_size, image_size, depth_size),
            condition_tensors=condition_tensors
        )

    @torch.no_grad()
    def interpolate(self, x1, x2, t=None, lam=0.5):
        b, *_, device = *x1.shape, x1.device
        t = default(t, self.num_timesteps - 1)
        assert x1.shape == x2.shape

        t_batched = torch.stack([torch.tensor(t, device=device)] * b)
        xt1, xt2 = map(lambda x: self.q_sample(x, t=t_batched), (x1, x2))

        img = (1 - lam) * xt1 + lam * xt2
        for i in tqdm(reversed(range(0, t)), desc='interpolation sample time step', total=t):
            img = self.p_sample(img, torch.full((b,), i, device=device, dtype=torch.long))
        return img

    def q_sample(self, x_start, t, noise=None, c=None):
        noise = default(noise, lambda: torch.randn_like(x_start))
        x_hat = 0.
        return (
            extract(self.sqrt_alphas_cumprod, t, x_start.shape) * x_start +
            extract(self.sqrt_one_minus_alphas_cumprod, t, x_start.shape) * noise +
            x_hat
        )

    def p_losses(self, x_start, t, condition_tensors=None, noise=None):
        b, c, h, w, d = x_start.shape
        noise = default(noise, lambda: torch.randn_like(x_start))

        if self.with_condition:
            x_noisy = self.q_sample(x_start, t=t, noise=noise)
            x_recon = self.denoise_fn(
                torch.cat([x_noisy, condition_tensors], 1), t
            )
        else:
            x_noisy = self.q_sample(x_start, t=t, noise=noise)
            x_recon = self.denoise_fn(x_noisy, t)

        if self.loss_type == 'l1':
            loss = (noise - x_recon).abs().mean()
        elif self.loss_type == 'l2':
            loss = F.mse_loss(x_recon, noise)
        else:
            raise NotImplementedError()

        return loss

    def forward(self, x, condition_tensors=None, *args, **kwargs):
        b, c, h, w, d, device = *x.shape, x.device
        assert h == self.image_size and w == self.image_size and d == self.depth_size, \
            f'Expected (H,W,D)=({self.image_size},{self.image_size},{self.depth_size}), got ({h},{w},{d}).'
        t = torch.randint(0, self.num_timesteps, (b,), device=device).long()
        return self.p_losses(x, t, condition_tensors=condition_tensors, *args, **kwargs)

class Trainer(object):
    def __init__(
        self,
        diffusion_model,
        train_loader,
        val_dataset=None,
        ema_decay=0.995,
        image_size=128,
        depth_size=128,
        train_batch_size=1,
        train_lr=2e-6,
        train_num_steps=100000,
        gradient_accumulate_every=2,
        fp16=False,
        step_start_ema=2000,
        update_ema_every=10,
        save_and_sample_every=1000,
        results_folder='./results',
        with_condition=False,
        with_pairwised=False
    ):
        super().__init__()
        self.model = diffusion_model
        self.ema = EMA(ema_decay)
        self.ema_model = copy.deepcopy(self.model)
        self.update_ema_every = update_ema_every

        self.step_start_ema = step_start_ema
        self.save_and_sample_every = save_and_sample_every

        self.batch_size = train_batch_size
        self.image_size = diffusion_model.image_size
        self.depth_size = diffusion_model.depth_size
        self.gradient_accumulate_every = gradient_accumulate_every
        self.train_num_steps = train_num_steps

        self.ds = train_loader.dataset
        self.dl = cycle(train_loader)
        self.opt = Adam(diffusion_model.parameters(), lr=train_lr)
        self.scheduler = StepLR(self.opt, step_size=125000, gamma=0.5)
        self.train_lr = train_lr
        self.train_batch_size = train_batch_size
        self.with_condition = with_condition

        self.step = 0

        self.fp16 = fp16
        if fp16 and APEX_AVAILABLE:
            (self.model, self.ema_model), self.opt = amp.initialize([self.model, self.ema_model], self.opt, opt_level='O1')

        self.results_folder = Path(results_folder)
        self.results_folder.mkdir(exist_ok=True)

        # Create modality-specific directories without altering orientation
        export_dirs = ["t1n", "t1c", "t2w", "t2f"]
        for dir_name in export_dirs:
            os.makedirs(self.results_folder / dir_name, exist_ok=True)
        os.makedirs(self.results_folder / "model", exist_ok=True)

        self.log_dir = self.create_log_dir()
        self.writer = SummaryWriter(log_dir=self.log_dir)
        self.reset_parameters()

    def create_log_dir(self):
        now = datetime.datetime.now().strftime("%y-%m-%dT%H%M%S")
        log_dir = os.path.join(".logs", now)
        os.makedirs(log_dir, exist_ok=True)
        return log_dir

    def reset_parameters(self):
        self.ema_model.load_state_dict(self.model.state_dict())

    def step_ema(self):
        if self.step < self.step_start_ema:
            self.reset_parameters()
            return
        self.ema.update_model_average(self.ema_model, self.model)

    def save(self, milestone):
        data = {
            'step': self.step,
            'model': self.model.state_dict(),
            'ema': self.ema_model.state_dict()
        }
        torch.save(data, str(self.results_folder / f'model/model-{milestone}.pt'))

    def load(self, milestone):
        data = torch.load(str(self.results_folder / f'model/model-{milestone}.pt'))
        self.step = data['step']
        self.model.load_state_dict(data['model'])
        self.ema_model.load_state_dict(data['ema'])

    def train(self):
        backwards = partial(loss_backwards, self.fp16)
        start_time = time.time()

        while self.step < self.train_num_steps:
            accumulated_loss = []
            for i in range(self.gradient_accumulate_every):
                if self.with_condition:
                    data = next(self.dl)
                    input_tensors = data['input'].cuda()    # [B, 6, H, W, D]
                    target_tensors = data['target'].cuda()  # [B, 4, H, W, D]
                    affines = data['affine']                 # [B, 4, 4]

                    # Ensure that the shapes match
                    b, c_in, h, w, d = input_tensors.shape
                    b_out, c_out, h_out, w_out, d_out = target_tensors.shape

                    if h != self.image_size or w != self.image_size or d != self.depth_size:
                        raise ValueError(
                            f"[ERROR] Input dims mismatch: got ({b},{c_in},{h},{w},{d}). Expected ({b},{c_in},{self.image_size},{self.image_size},{self.depth_size})."
                        )
                    if h_out != self.image_size or w_out != self.image_size or d_out != self.depth_size:
                        raise ValueError(
                            f"[ERROR] Target dims mismatch: got ({b_out},{c_out},{h_out},{w_out},{d_out}). Expected ({b_out},{c_out},{self.image_size},{self.image_size},{self.depth_size})."
                        )

                    # Pass tensors as [B, C, H, W, D] directly
                    loss = self.model(target_tensors, condition_tensors=input_tensors)

                    # Store affines for saving generated images
                    current_affines = affines
                else:
                    data = next(self.dl).cuda()
                    loss = self.model(data)

                loss = loss.sum() / self.batch_size
                print(f"[INFO] Step {self.step}: loss = {loss.item()}")
                logging.info(f"Step {self.step}: loss = {loss.item()}")
                backwards(loss / self.gradient_accumulate_every, self.opt)
                accumulated_loss.append(loss.item())

            average_loss = np.mean(accumulated_loss)
            self.writer.add_scalar("training_loss", average_loss, self.step)

            self.opt.step()
            self.scheduler.step()
            self.opt.zero_grad()

            if self.step % self.update_ema_every == 0:
                self.step_ema()

            if self.step != 0 and self.step % self.save_and_sample_every == 0:
                milestone = self.step // self.save_and_sample_every
                batches = num_to_groups(1, self.batch_size)

                if self.with_condition:
                    device = next(self.ema_model.parameters()).device
                    all_images_list = list(map(
                        lambda n: self.ema_model.sample(
                            batch_size=n,
                            condition_tensors=(
                                sample_conditions(self.ds, batch_size=n)
                                .to(device)
                            )
                        ),
                        batches
                    ))
                else:
                    all_images_list = list(map(
                        lambda n: self.ema_model.sample(batch_size=n),
                        batches
                    ))

                all_images = torch.cat(all_images_list, dim=0)

                if self.with_condition:
                    # Assuming batch size is 1, replicate affines accordingly
                    # If batch size >1, adjust as needed
                    affines_to_use = current_affines
                else:
                    affines_to_use = None  # No affines to pass

                self.save_generated_images(all_images, affines_to_use, milestone)
                self.save(milestone)

            self.step += 1

        print('training completed')
        self.finalize_training(start_time)

    def save_generated_images(self, all_images, affines, milestone):
        """
        Save generated images as NIfTI files using corresponding affines.
        Assuming all_images shape: [B, C, H, W, D].
        """
        t1n_images = all_images[:, 0, ...]  # [B, H, W, D]
        t1c_images = all_images[:, 1, ...]
        t2w_images = all_images[:, 2, ...]
        t2f_images = all_images[:, 3, ...]

        # If affines are provided, use them; else, use identity
        for modality_images, modality in zip(
            [t1n_images, t1c_images, t2w_images, t2f_images],
            ['t1n', 't1c', 't2w', 't2f']
        ):
            for b in range(modality_images.shape[0]):
                modality_image = modality_images[b].cpu().numpy()  # shape [H, W, D]
                if affines is not None:
                    current_affine = affines[b].cpu().numpy()
                else:
                    current_affine = np.eye(4)

                nifti_img = nib.Nifti1Image(modality_image, affine=current_affine)
                save_path = self.results_folder / f'{modality}/sample-{milestone}_batch-{b}.nii.gz'
                nib.save(nifti_img, str(save_path))

    def finalize_training(self, start_time):
        end_time = time.time()
        execution_time = (end_time - start_time) / 3600.0
        print(f"[INFO] Training completed in {execution_time:.2f} hours")
        self.writer.add_hparams(
            {
                "lr": self.train_lr,
                "batchsize": self.train_batch_size,
                "image_size": self.image_size,
                "depth_size": self.depth_size,
                "execution_time_hours": execution_time
            },
            {"last_loss": self.writer.scalar_dict["training_loss"]}
        )
        self.writer.close()

def random_flip_3d(volume):
    """
    Removed: Flipping is omitted to preserve scan orientation.

    # Flipping is intentionally omitted to maintain orientation.

def random_affine_3d(volume, scale_range=(0.95, 1.05), shift_range=(-2, 2)):
    """
    Applies a small uniform scale and translation to the entire 3D volume.
    Preserves anatomical orientation by avoiding rotations and non-uniform scaling.
    Returns the transformed volume and a description of the affine transformation.
    """
    scale_factor = np.random.uniform(*scale_range)
    shift_z = np.random.randint(*shift_range)
    shift_y = np.random.randint(*shift_range)
    shift_x = np.random.randint(*shift_range)

    D, H, W = volume.shape
    center_z = D / 2.0
    center_y = H / 2.0
    center_x = W / 2.0

    # Uniform scaling matrix (no rotation or reflection)
    scale_matrix = np.diag([1/scale_factor, 1/scale_factor, 1/scale_factor])

    # Calculate offset to keep the scaling centered
    offset_z = center_z - scale_factor * center_z + shift_z
    offset_y = center_y - scale_factor * center_y + shift_y
    offset_x = center_x - scale_factor * center_x + shift_x
    offset = np.array([offset_z, offset_y, offset_x])

    transformed = affine_transform(
        volume,
        scale_matrix,
        offset=offset,
        order=0,  # nearest-neighbor to preserve labels
        mode='constant',
        cval=0
    )

    # Ensure the transformed volume is of integer type
    transformed = transformed.astype(np.uint8)

    affine_desc = f"Scale: {scale_factor:.3f}, Shifts: (z:{shift_z}, y:{shift_y}, x:{shift_x})"
    print(f" [Affine] {affine_desc}")
    return transformed, affine_desc

def random_small_tumor_displacement(volume, max_shift=1):
    """
    Slightly moves the tumor (labels >=2) within the head (label=1).
    Ensures that the displacement keeps the tumor within head boundaries.
    Returns the transformed volume and a description of the displacement.
    """
    # Identify all tumor labels (assuming labels >=2 are tumor)
    tumor_labels = [label for label in np.unique(volume) if label >= 2]
    if not tumor_labels:
        print(" [TumorDisplacement] No tumor found. Skipping displacement.")
        return volume, "Tumor Shift: None"

    # Create a mask for all tumor labels
    tumor_mask = np.isin(volume, tumor_labels)

    tumor_coords = np.argwhere(tumor_mask)
    if len(tumor_coords) == 0:
        print(" [TumorDisplacement] No tumor voxels found. Skipping displacement.")
        return volume, "Tumor Shift: None"

    D, H, W = volume.shape

    # Calculate centroid of the tumor for better shift targeting
    centroid = tumor_coords.mean(axis=0).astype(int)

    # Define the region of interest (ROI) around the centroid
    roi_size = 10  # Adjust based on your data
    z_start = max(centroid[0] - roi_size, 0)
    z_end = min(centroid[0] + roi_size, D)
    y_start = max(centroid[1] - roi_size, 0)
    y_end = min(centroid[1] + roi_size, H)
    x_start = max(centroid[2] - roi_size, 0)
    x_end = min(centroid[2] + roi_size, W)

    # Extract the ROI
    roi = volume[z_start:z_end, y_start:y_end, x_start:x_end]

    # Adjust tumor coordinates relative to ROI
    relative_tumor_coords = tumor_coords - np.array([z_start, y_start, x_start])

    # Attempt multiple shifts to find a valid displacement
    NUM_ATTEMPTS = 30  # Increased number of attempts
    for attempt in range(NUM_ATTEMPTS):
        dz = np.random.randint(-max_shift, max_shift + 1)
        dy = np.random.randint(-max_shift, max_shift + 1)
        dx = np.random.randint(-max_shift, max_shift + 1)

        # Skip zero shift
        if dz == 0 and dy == 0 and dx == 0:
            continue

        # Shifted coordinates within ROI
        shifted_coords = relative_tumor_coords + np.array([dz, dy, dx])

        # Check if shifted coordinates are within ROI bounds
        if (shifted_coords[:, 0].min() < 0 or shifted_coords[:, 0].max() >= (z_end - z_start) or
            shifted_coords[:, 1].min() < 0 or shifted_coords[:, 1].max() >= (y_end - y_start) or
            shifted_coords[:, 2].min() < 0 or shifted_coords[:, 2].max() >= (x_end - x_start)):
            continue  # Shifted tumor goes out of ROI bounds

        # Check if all shifted tumor voxels are within the head (labels >=1)
        head_region = roi[shifted_coords[:, 0], shifted_coords[:, 1], shifted_coords[:, 2]]
        if np.all(head_region >= 1):
            # Shift is valid
            out_vol = volume.copy()

            # Extract labels of the original tumor voxels
            original_labels = out_vol[tumor_coords[:, 0], tumor_coords[:, 1], tumor_coords[:, 2]]

            # Set original tumor voxels to head label (assuming label=1)
            # If the original labels have multiple classes, preserve their labels
            # Here, we assume head label is 1
            # If head label varies, adjust accordingly
            out_vol[tumor_coords[:, 0], tumor_coords[:, 1], tumor_coords[:, 2]] = 1

            # Calculate absolute shifted coordinates
            absolute_shifted_coords = shifted_coords + np.array([z_start, y_start, x_start])

            # Extract labels of the shifted tumor voxels (from original positions)
            shifted_labels = original_labels  # Preserve original labels

            # Set new tumor voxels with their original labels
            out_vol[absolute_shifted_coords[:, 0],
                    absolute_shifted_coords[:, 1],
                    absolute_shifted_coords[:, 2]] = shifted_labels

            displacement_desc = f"Tumor Shift: (dz:{dz}, dy:{dy}, dx:{dx})"
            print(f" [TumorDisplacement] {displacement_desc}")
            return out_vol, displacement_desc

    # If no valid shift found after attempts, return unchanged
    print(" [TumorDisplacement] No valid shift found after multiple attempts. Skipping displacement.")
    return volume, "Tumor Shift: None"

def random_augment(volume):
    """
    Augmentation pipeline:
      1) Apply small uniform scaling and shifting.
      2) Slightly displace the tumor within the head.
    Returns the augmented volume and a description of all transformations.
    """
    out = volume.copy()
    transformation_details = []

    # 1) Small uniform scale + translation
    out, affine_desc = random_affine_3d(out, scale_range=(0.98, 1.02), shift_range=(-1, 1))
    transformation_details.append(affine_desc)

    # 2) Slightly move the tumor
    out, displacement_desc = random_small_tumor_displacement(out, max_shift=1)
    transformation_details.append(displacement_desc)

    # **Ensure the final augmented volume is of integer type and clip labels**
    out = out.astype(np.uint8)
    out = np.clip(out, 0, 5)  # Adjusted to accommodate labels up to 5

    # Combine all transformation descriptions
    full_transformation_desc = "; ".join(transformation_details)
    return out, full_transformation_desc
